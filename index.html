# 人马流派配兵效率估算（简化版）
def hog_efficiency(hogs, heals, bomb_lv):
  base = hogs * 62 # 单猪期望输出
  heal_bonus = heals * 180 * (1 - bomb_lv/20)
  threat = base + heal_bonus - bomb_lv * 45
  return round(threat / 1000, 2)

print(hog_efficiency(24, 5, 8)) # => 15.36
